get_units_digit = lambda a, b: pow(a, b, 10)

# User Input
base = int(input("Enter base (a): "))
exponent = int(input("Enter exponent (b): "))

# Output
print(get_units_digit(base, exponent))

num = int(input("Enter a number: "))

# Extract unique digits, ignoring zero to avoid a ZeroDivisionError
unique_digits = {int(d) for d in str(num) if d != '0'}

# Count how many digits divide the original number perfectly
count = sum(1 for d in unique_digits if num % d == 0)

print(f"Count of perfect divisors: {count}")

import math

num = int(input("Enter a large number: "))
divisors = []

# Loop only up to the square root of the number
for i in range(1, int(math.isqrt(num)) + 1):
    if num % i == 0:
        divisors.append(i)          # Small factor
        if i != num // i:           # Avoid duplicates for perfect squares (like 4*4=16)
            divisors.append(num // i) # Large matching factor

divisors.sort()
print(f"Optimized Divisors of {num}: {divisors}")
